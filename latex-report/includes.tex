% --- CUSTOM CHAPTER HEADING OVERRIDES (SMU MAROON THEME) ---
\makeatletter
\def\@makechapterhead#1{%
  {\parindent \z@ \centering
    \interlinepenalty\@M
    \fontsize{16}{19}\selectfont \bfseries \textcolor{smubrand}{Chapter \thechapter.\enspace #1}\par\nobreak
    \vskip 8\p@
  }}

\def\@makeschapterhead#1{%
  {\parindent \z@ \centering
    \interlinepenalty\@M
    \fontsize{16}{19}\selectfont \bfseries \textcolor{smubrand}{#1}\par\nobreak
    \vskip 8\p@
  }}

% Remove the forced new page before \chapter in report class
\patchcmd{\chapter}{\if@openright\cleardoublepage\else\clearpage\fi}{}{}{}

% Horizontal rule before chapter title, only when NOT at top of page
\newif\iffirstchapter
\firstchaptertrue
\preto\chapter{%
  \iffirstchapter
    \global\firstchapterfalse
  \else
    % \ifdim measures how far down the page we are;
    % if \pagetotal > 1pt we're mid-page, so draw the rule
    \ifdim\pagetotal>1pt
      \par\vspace{0.5em}%
      \noindent\rule{\textwidth}{0.4pt}%
      \par\vspace{0.5em}%
    \fi
  \fi
}
