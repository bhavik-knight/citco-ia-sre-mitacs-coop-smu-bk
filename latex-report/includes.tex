% Shared helper — 16pt baseline, bold, maroon, centered, ALL CAPS
% Neeyati Heading 1: before=2pt, after=12pt
% Space before chapter should account for parskip from previous paragraph
\def\@chapterheadingline#1{%
  \vskip 4\p@  % Space before chapter heading
  {\parindent \z@ \centering
   \interlinepenalty\@M
   \fontsize{16}{16}\selectfont \bfseries \textcolor{smubrand}{#1}\par\nobreak
   \vskip 2\p@}%  % Space after chapter heading
}

% Numbered chapters: "1. CHAPTER TITLE"
\def\@makechapterhead#1{%
  \@chapterheadingline{\thechapter.\enspace \MakeUppercase{#1}}%
}

% Unnumbered chapters (Acknowledgement, TOC, etc.): "CHAPTER TITLE"
\def\@makeschapterhead#1{%
  \@chapterheadingline{\MakeUppercase{#1}}%
}

\makeatother
