% ==============================================================================
% 4. HEADINGS
% ==============================================================================
% Chapter, section, subsection, subsubsection styling
% Font hierarchy: Chapter=16pt, Section=14pt, Subsection=12pt, Subsubsection=11pt
% All bold maroon, consistent 4pt before / ~0pt (0.001pt) after spacing
%
% IMPORTANT: All headings use \needspace to prevent orphaned headings at page bottom
% This ensures at least 4 lines of content follow each heading on the same page

\usepackage{titlesec}
\usepackage{etoolbox}
\usepackage{needspace}   % Prevent orphaned headings at page bottom

\makeatletter

% --- Section (Level 1) — 14pt bold maroon ---
% \Needspace* is strict: if not enough room, forces page break
% 36pt = heading height + 1 line of content minimum
\renewcommand\section{\Needspace*{36pt}\@startsection{section}{1}{\z@}%
  {-12pt \@plus -2pt \@minus -2pt}%  % before: negative = no indent after
  {6pt \@plus 2pt}%                   % after: positive = keep with following text
  {\fontsize{14}{14}\selectfont\bfseries\color{smubrand}}}

% --- Subsection (Level 2) — 12pt bold maroon ---
% 32pt = heading + 1 line minimum
\renewcommand\subsection{\Needspace*{32pt}\@startsection{subsection}{2}{\z@}%
  {-10pt \@plus -2pt \@minus -2pt}%
  {4pt \@plus 2pt}%
  {\fontsize{12}{12}\selectfont\fontseries{b}\selectfont\color{smubrand}}}

% --- Subsubsection (Level 3) — 11pt bold maroon ---
% 28pt = heading + 1 line minimum
\renewcommand\subsubsection{\Needspace*{28pt}\@startsection{subsubsection}{3}{\z@}%
  {-8pt \@plus -2pt \@minus -2pt}%
  {2pt \@plus 1pt}%
  {\fontsize{12}{12}\selectfont\fontseries{b}\selectfont\color{smubrand}}}

% --- Chapter Heading — 16pt bold maroon, centered, ALL CAPS ---
% Remove \clearpage from chapter command — chapters continue on same page
\patchcmd{\chapter}{\clearpage}{}{}{}
\patchcmd{\chapter}{\if@openright\cleardoublepage\else\clearpage\fi}{}{}{}

% Shared helper for chapter headings
% Spacing: before/after for visual separation
% \Needspace*{40pt} ensures room for heading + at least 1 line of content
\def\@chapterheadingline#1{%
  \Needspace*{40pt}%  % Strict: heading (~20pt) + minimum 1 line content (~20pt)
  \vskip 12\p@ \@plus 2\p@ \@minus 2\p@  % Space before chapter heading
  {\parindent \z@ \centering
   \interlinepenalty\@M
   \fontsize{16}{16}\selectfont \bfseries \textcolor{smubrand}{#1}\par\nobreak
   \vskip 6\p@ \@plus 2\p@}%  % Space after chapter heading
}

% Numbered chapters: "1. CHAPTER TITLE"
\def\@makechapterhead#1{%
  \@chapterheadingline{\thechapter.\enspace \MakeUppercase{#1}}%
}

% Unnumbered chapters (Acknowledgement, TOC, etc.): "CHAPTER TITLE"
\def\@makeschapterhead#1{%
  \@chapterheadingline{\MakeUppercase{#1}}%
}

\makeatother
