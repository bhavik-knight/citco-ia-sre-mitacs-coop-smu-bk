WITH filtered_items AS (
    SELECT
        item_id,
        corr_acc_no,
        sfield_8,
        value_date,
        date_11,
        flag_2,
        last_action_date,
        date_25
    FROM TLM_SCHEMA.item
    WHERE flag_46 = 4
      AND ls_type IN (9107, 9108, 9126, 9127)
      AND sfield_8 IS NOT NULL
      AND sfield_8 NOT IN (' ', '0')
      AND date_11 IS NOT NULL
),
filtered_banks AS (
    SELECT corr_acc_no
    FROM TLM_SCHEMA.bank
    WHERE agent_code = :agentcode
      AND string01 IN {funds_clause}
),
filtered_proofs AS (
    SELECT
        proof_id,
        corr_acc_no,
        proof_created_datetime,
        proof_datetime
    FROM TLM_SCHEMA.recs_bdr_proof
    WHERE proof_entity_type = 8124
      AND overall_status = 1
      AND proof_count9 = :process
      AND proof_datetime = :proofdate
)
SELECT
    MAX(CASE WHEN i.sfield_8 LIKE '%' || CHR(64) || '%'
             THEN SUBSTR(i.sfield_8, 0, INSTR(i.sfield_8, '@') - 1)
             ELSE i.sfield_8 END) AS "TID",
    TO_CHAR(MIN(CASE WHEN i.date_11 = '1-JAN-1900'
                     THEN i.value_date
                     ELSE i.date_11 END), 'MM/DD/YYYY') AS "Trade Date",
    MIN(CASE WHEN i.date_11 = '1-JAN-1900'
             THEN i.value_date
             ELSE i.date_11 END) AS "Date_Sort"
FROM filtered_items i
INNER JOIN filtered_banks b ON b.corr_acc_no = i.corr_acc_no
INNER JOIN TLM_SCHEMA.tlm_bdr_associated_link ritbal
    ON ritbal.linked_entity_key = i.item_id
   AND ritbal.linked_entity_type = 8018
LEFT OUTER JOIN filtered_proofs proof
    ON proof.proof_id = ritbal.entity_key
   AND ritbal.entity_type = 8016
WHERE ((i.flag_2 = '0')
   OR (i.flag_2 IN ('1', '3')
       AND (i.last_action_date > proof.proof_created_datetime
            OR i.date_25 > proof.proof_datetime)))
GROUP BY i.sfield_8
ORDER BY "Date_Sort" ASC
