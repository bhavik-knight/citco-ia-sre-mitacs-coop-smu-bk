#!/bin/bash
# Script to get resources from all regions
regions=$(aws ec2 describe-regions --query 'Regions[].RegionName' --output text)
echo '{"resources": [' > all_resources.json

first=true
for region in $regions; do
    echo "Checking region: $region"
    resources=$(aws resourcegroupstaggingapi get-resources \
        --region $region \
        --tag-filters Key=budget,Values=rpacfs1,ia-bpo \
        --output json)
    
    if [ "$first" = true ]; then
        echo "$resources" | jq -r '.ResourceTagMappingList[]' >> all_resources.json
        first=false
    else
        echo "," >> all_resources.json
        echo "$resources" | jq -r '.ResourceTagMappingList[]' >> all_resources.json
    fi
done

echo ']}' >> all_resources.json
